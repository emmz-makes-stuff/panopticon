# Logo assets: how they were measured and made

## 1.1 What the source measured

`docs/images/logo.webp` is 1024×1024 with no alpha (`sips -g hasAlpha` → `no`). Measured by `measure.mjs` (below) on
`dwebp -quiet docs/images/logo.webp -ppm -o src.ppm`:

- **Inset: none.** The rounded square is full-bleed. Along rows and columns at mid-height, the first light-to-dark
  crossing is the artwork, not a margin.
- **Corner:** a smooth curve, not a circular arc. In the top-left, the light/dark edge is at x = 165.6 on row 0 and
  x = 53.6 on row 50, and reaches x = 0 by row ~170. The diagonal crossing is at 51.66, which is close to a circle of
  r ≈ 170–176 but fits neither exactly. So the mask comes from the pixels themselves (design D1, updated).
- **Background:** `#121928`/`#121929` (top and bottom patches), taken as `#121929`. Outside the corners the pixels are
  `#FEFEFE`/`#FCFFFE`.
- **Background texture:** outside the artwork, the max channel deviation from `#121929` is 0 for 73.2% of pixels, 1–2
  for 24.7%, 3–4 for 2.0% and 5–7 for 0.14%. In the social preview it showed as ghostly blocks around the triangle.
  The glow under the triangle's bottom edge falls 230 → 22 → 31 → 21 → 18 → 11 → 5 → 4 → 2 over 5 px steps. Hence the
  smoothstep from 3 to 8.

## 1.3 How to make them again

From the repo root, with `dwebp` (Homebrew `webp`) and Node:

```sh
mkdir -p "$TMPDIR/logo" && cp ppm.mjs make.mjs check.mjs "$TMPDIR/logo/"   # the three scripts below
dwebp -quiet docs/images/logo.webp -ppm -o "$TMPDIR/logo/src.ppm"
node "$TMPDIR/logo/make.mjs" "$TMPDIR/logo/src.ppm" "$PWD"
node "$TMPDIR/logo/check.mjs" web/public/favicon-32.png web/public/apple-touch-icon.png web/public/logo.png docs/images/logo.png docs/images/social-preview.png
```

The constants in `make.mjs` (`BG`, `CORNER`, the 3–8 cut-off) are specific to this source. Measure a new source
with `measure.mjs` first.

### ppm.mjs

```js
import fs from "node:fs";
export function readPpm(path) {
  const b = fs.readFileSync(path); let p = 0; const f = [];
  while (f.length < 4) { while (/\s/.test(String.fromCharCode(b[p]))) p++; let s = ""; while (!/\s/.test(String.fromCharCode(b[p]))) s += String.fromCharCode(b[p++]); f.push(s); }
  p++; const w = +f[1], h = +f[2];
  return { w, h, px: (x, y) => { const i = p + 3 * (y * w + x); return [b[i], b[i + 1], b[i + 2]]; } };
}
export const lum = ([r, g, b]) => 0.2126 * r + 0.7152 * g + 0.0722 * b;
```

### make.mjs

```js
// Makes Panopticon's PNG logo assets from docs/images/logo.webp (decoded first with `dwebp -ppm`).
// Usage: node make.mjs <src.ppm> <repo-root>
import fs from "node:fs";
import path from "node:path";
import zlib from "node:zlib";
import { readPpm, lum } from "./ppm.mjs";

const [srcPath, root] = process.argv.slice(2);
const src = readPpm(srcPath);
const { w, h } = src;
const BG = [0x12, 0x19, 0x29]; // the rounded square's background (measured)
const WHITE_L = 254, BG_L = lum(BG);
const CORNER = 180; // corner zones: the curve ends by ~170 px, and the artwork starts beyond 180 px

const toLin = (c) => { c /= 255; return c <= 0.04045 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4; };
const toSrgb = (l) => { const c = l <= 0.0031308 ? l * 12.92 : 1.055 * l ** (1 / 2.4) - 0.055; return Math.max(0, Math.min(255, Math.round(c * 255))); };

// Source as premultiplied linear RGBA floats.
const lin = new Float64Array(w * h * 4);
for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) {
  const c = src.px(x, y);
  const inCorner = (x < CORNER || x >= w - CORNER) && (y < CORNER || y >= h - CORNER);
  let a = 1, rgb = c;
  if (!inCorner) {
    // Flatten the source's faint background texture (max channel deviation <= 7) to the exact background, keeping the
    // glow: a soft cut-off from 3 (gone) to 8 (untouched).
    const d = Math.max(...c.map((v, k) => Math.abs(v - BG[k])));
    const t = Math.min(1, Math.max(0, (d - 3) / 5)), s = t * t * (3 - 2 * t);
    rgb = c.map((v, k) => BG[k] + (v - BG[k]) * s);
  }
  if (inCorner) { a = Math.max(0, Math.min(1, (WHITE_L - lum(c)) / (WHITE_L - BG_L))); rgb = BG; }
  const i = 4 * (y * w + x);
  for (let k = 0; k < 3; k++) lin[i + k] = toLin(rgb[k]) * a;
  lin[i + 3] = a;
}

// Area-average downsample of premultiplied linear RGBA to n×n.
function resize(n) {
  const out = new Float64Array(n * n * 4), s = w / n;
  for (let dy = 0; dy < n; dy++) for (let dx = 0; dx < n; dx++) {
    const x0 = dx * s, x1 = x0 + s, y0 = dy * s, y1 = y0 + s; const acc = [0, 0, 0, 0]; let wt = 0;
    for (let y = Math.floor(y0); y < Math.ceil(y1); y++) {
      const wy = Math.min(y + 1, y1) - Math.max(y, y0);
      for (let x = Math.floor(x0); x < Math.ceil(x1); x++) {
        const wx = Math.min(x + 1, x1) - Math.max(x, x0), ww = wx * wy, i = 4 * (y * w + x);
        for (let k = 0; k < 4; k++) acc[k] += lin[i + k] * ww; wt += ww;
      }
    }
    const o = 4 * (dy * n + dx); for (let k = 0; k < 4; k++) out[o + k] = acc[k] / wt;
  }
  return { n, data: out };
}

// Premultiplied linear → 8-bit sRGB rows. opaque: composite over BG and drop alpha.
function rows(img, width, height, opaque) {
  const ch = opaque ? 3 : 4, buf = Buffer.alloc(height * (1 + width * ch)); const bgLin = BG.map(toLin);
  for (let y = 0; y < height; y++) {
    buf[y * (1 + width * ch)] = 0;
    for (let x = 0; x < width; x++) {
      const i = 4 * (y * width + x), a = img[i + 3], o = y * (1 + width * ch) + 1 + x * ch;
      for (let k = 0; k < 3; k++) {
        const c = opaque ? img[i + k] + bgLin[k] * (1 - a) : a > 0 ? img[i + k] / a : 0;
        buf[o + k] = toSrgb(c);
      }
      if (!opaque) buf[o + 3] = Math.round(a * 255);
    }
  }
  return buf;
}

const crcTable = Array.from({ length: 256 }, (_, n) => { let c = n; for (let k = 0; k < 8; k++) c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1; return c >>> 0; });
const crc32 = (b) => { let c = 0xffffffff; for (const x of b) c = crcTable[(c ^ x) & 0xff] ^ (c >>> 8); return (c ^ 0xffffffff) >>> 0; };
function chunk(type, data) { const len = Buffer.alloc(4); len.writeUInt32BE(data.length); const td = Buffer.concat([Buffer.from(type), data]); const crc = Buffer.alloc(4); crc.writeUInt32BE(crc32(td)); return Buffer.concat([len, td, crc]); }
function png(file, img, width, height, opaque) {
  const ihdr = Buffer.alloc(13); ihdr.writeUInt32BE(width, 0); ihdr.writeUInt32BE(height, 4); ihdr[8] = 8; ihdr[9] = opaque ? 2 : 6;
  const idat = zlib.deflateSync(rows(img, width, height, opaque), { level: 9 });
  const out = Buffer.concat([Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]), chunk("IHDR", ihdr), chunk("IDAT", idat), chunk("IEND", Buffer.alloc(0))]);
  fs.mkdirSync(path.dirname(file), { recursive: true }); fs.writeFileSync(file, out);
  console.log(`${path.relative(root, file)} ${width}x${height} ${opaque ? "RGB" : "RGBA"} ${out.length} bytes`);
}

const at = (p) => path.join(root, p);
for (const [file, n, opaque] of [["web/public/favicon-32.png", 32, false], ["web/public/apple-touch-icon.png", 180, true], ["web/public/logo.png", 128, false], ["docs/images/logo.png", 512, false]]) {
  const r = resize(n); png(at(file), r.data, n, n, opaque);
}

// Social preview: the 400 px logo centred on a 1280×640 field of the logo's background.
{
  const W = 1280, H = 640, L = 400, r = resize(L), bgLin = BG.map(toLin), img = new Float64Array(W * H * 4);
  const ox = (W - L) / 2, oy = (H - L) / 2;
  for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) {
    const o = 4 * (y * W + x); let s = [0, 0, 0, 0];
    if (x >= ox && x < ox + L && y >= oy && y < oy + L) { const i = 4 * ((y - oy) * L + (x - ox)); s = [r.data[i], r.data[i + 1], r.data[i + 2], r.data[i + 3]]; }
    for (let k = 0; k < 3; k++) img[o + k] = s[k] + bgLin[k] * (1 - s[3]);
    img[o + 3] = 1;
  }
  png(at("docs/images/social-preview.png"), img, W, H, true);
}
```

### check.mjs

```js
import fs from "node:fs"; import zlib from "node:zlib";
for (const f of process.argv.slice(2)) {
  const b = fs.readFileSync(f); let p = 8, w, h, t, idat = [];
  while (p < b.length) { const len = b.readUInt32BE(p), type = b.toString("ascii", p + 4, p + 8), d = b.subarray(p + 8, p + 8 + len);
    if (type === "IHDR") { w = d.readUInt32BE(0); h = d.readUInt32BE(4); t = d[9]; } if (type === "IDAT") idat.push(d); p += 12 + len; }
  const raw = zlib.inflateSync(Buffer.concat(idat)), ch = t === 6 ? 4 : 3, row = 1 + w * ch;
  const px = (x, y) => [...raw.subarray(y * row + 1 + x * ch, y * row + 1 + x * ch + ch)];
  let transparent = 0; if (ch === 4) for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) if (px(x, y)[3] < 255) transparent++;
  // brightest near-edge pixel with alpha>0 in the corner (white fringe check)
  let fringe = 0; if (ch === 4) for (let y = 0; y < h / 4; y++) for (let x = 0; x < w / 4; x++) { const c = px(x, y); if (c[3] > 0 && c[3] < 255) fringe = Math.max(fringe, c[0], c[1], c[2]); }
  console.log(f, `corner ${px(0, 0)} centre ${px(w >> 1, h >> 1)} non-opaque ${transparent} max-edge-rgb ${fringe}`);
}
```

### measure.mjs

```js
import { readPpm, lum } from "./ppm.mjs";
const img = readPpm(process.argv[2]); const { w, h, px } = img;
// first x (from the left) where luminance drops below the midpoint, sub-pixel by linear interpolation
const mid = 128;
function edgeX(y) { for (let x = 1; x < w / 2; x++) { const a = lum(px(x - 1, y)), b = lum(px(x, y)); if (a >= mid && b < mid) return x - 1 + (a - mid) / (a - b); } return null; }
function edgeY(x) { for (let y = 1; y < h / 2; y++) { const a = lum(px(x, y - 1)), b = lum(px(x, y)); if (a >= mid && b < mid) return y - 1 + (a - mid) / (a - b); } return null; }
console.log("size", w, h);
console.log("left edge at mid rows", [400, 512, 600].map(edgeX).map(v => v?.toFixed(2)));
console.log("top edge at mid cols", [400, 512, 600].map(edgeY).map(v => v?.toFixed(2)));
console.log("right edge (mirror)", (() => { const y = 512; for (let x = w - 2; x > w / 2; x--) { const a = lum(px(x + 1, y)), b = lum(px(x, y)); if (a >= mid && b < mid) return (x + 1 - (a - mid) / (a - b)).toFixed(2); } })());
console.log("bottom edge (mirror)", (() => { const x = 512; for (let y = h - 2; y > h / 2; y--) { const a = lum(px(x, y + 1)), b = lum(px(x, y)); if (a >= mid && b < mid) return (y + 1 - (a - mid) / (a - b)).toFixed(2); } })());
// corner profile: for rows in the top-left corner, edge x; compare to circle and to the straight edge
const rows = [];
for (let y = 0; y <= 260; y += 10) rows.push([y, edgeX(y)]);
console.log("top-left profile y:x", rows.map(([y, x]) => `${y}:${x?.toFixed(1)}`).join(" "));
// diagonal
for (let d = 0; d < 300; d++) { const a = lum(px(d, d)), b = lum(px(d + 1, d + 1)); if (a >= mid && b < mid) { console.log("diagonal crossing", (d + (a - mid) / (a - b)).toFixed(2)); break; } }
// background colour: average a patch inside away from artwork
let s = [0, 0, 0], n = 0; for (let y = 60; y < 120; y++) for (let x = 400; x < 620; x++) { const c = px(x, y); s = s.map((v, i) => v + c[i]); n++; }
console.log("bg top patch", s.map(v => Math.round(v / n).toString(16).padStart(2, "0")).join(""));
s = [0, 0, 0]; n = 0; for (let y = 880; y < 960; y++) for (let x = 400; x < 620; x++) { const c = px(x, y); s = s.map((v, i) => v + c[i]); n++; }
console.log("bg bottom patch", s.map(v => Math.round(v / n).toString(16).padStart(2, "0")).join(""));
console.log("outside corner px", px(2, 2), px(w - 3, h - 3));
```
